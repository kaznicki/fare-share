    ```typescript
    export interface ParticipantBill {

    }

    export interface BillSplitResult {
    ```

    Internal helper `distributeProportionally(totalCents: number, subtotals: number[]): number[]`:
    - If `subtotals.length === 0`, return `[]`
    - Compute `grandSubtotal = subtotals.reduce((a, b) => a + b, 0)`
    - If `grandSubtotal === 0`: equal split with remainder via `Math.floor(totalCents / subtotals.length)` and give 1 extra cent to first `totalCents % subtotals.length` participants
    - Otherwise: compute exact float shares `(s / grandSubtotal) * totalCents`, floor each, compute remainders, sort indices by descending remainder (tiebreak by ascending index), give 1 extra cent to top `leftover` indices

    Main export `billSplit(params)`:
    - Params: `{ items: Array<{ id: string; priceCents: number }>, claims: Record<string, string[]>, participants: string[], taxCents: number, tipCents: number, unclaimedHandling: 'split' | 'host', hostName: string }`
    - Step 1: Build per-participant subtotals from claimed items. Shared items: `Math.round(item.priceCents / claimants.length)` per claimant.
    - Step 2: Handle unclaimed items. If `'host'`: add full price to host's subtotal. If `'split'`: compute total unclaimed cost, distribute via `distributeProportionally()` across all participants based on their current subtotals.
    - Step 3: Distribute tax via `distributeProportionally(taxCents, subtotalValues)`
    - Step 4: Distribute tip via `distributeProportionally(tipCents, subtotalValues)`
    - Step 5: Return `{ participants: [...] }` with each entry having `name, subtotalCents, taxShareCents, tipShareCents, totalCents` where `totalCents = subtotalCents + taxShareCents + tipShareCents`

       ```typescript
       if (

         typeof (msg as any).unclaimedHandling === 'string' &&
         typeof (msg as any).participantName === 'string'
       ) {
         const senderName = ((msg as any).participantName as string).trim().slice(0, MAX_NAME_LEN)
         const session = sessionStore.get(sessionId!)
         if (!session) return

         // Only host can finalize
         if (senderName !== session.hostName) return

         // Idempotency guard — do not re-run if already finalized
         if (session.finalized) return

         const handling = (msg as any).unclaimedHandling === 'host' ? 'host' : 'split'

         const result = billSplit({
           items: session.items,
           claims: session.claims,
           participants: session.participants,
           taxCents: session.taxCents,
           tipCents: session.tipCents,
           unclaimedHandling: handling,
           hostName: session.hostName,
         })

         sessionStore.finalize(sessionId!, result)

         // Broadcast updated snapshot (now includes finalized: true and finalizedBill)
         const data = sessionStore.getData(sessionId!)
         if (data) {
           sessionStore.broadcast(sessionId!, { type: 'session-snapshot', data })
         }
         return
       }
       ```

    Note: The `participantName` in the finalize message is used solely for the host identity check (`senderName !== session.hostName`). This is the security mitigation for T-5-01 (elevation of privilege).

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client to WebSocket server | Any participant can send any message type; server must validate sender identity for privileged actions |
| Client to REST API | POST /api/sessions body is untrusted input; hostName must be validated |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-5-01 | Elevation of Privilege | server.ts finalize handler | mitigate | Validate `msg.participantName === session.hostName` before executing finalize; non-host messages silently ignored |
| T-5-02 | Tampering | server.ts finalize handler | mitigate | Coerce `unclaimedHandling` to 'split' if value is not 'host': `const handling = v === 'host' ? 'host' : 'split'` |
| T-5-03 | Tampering | server.ts finalize handler | mitigate | Idempotency guard: `if (session.finalized) return` prevents double-finalization from overwriting stored results |
| T-5-04 | Tampering | app/api/sessions/route.ts | mitigate | Validate hostName with Zod: `z.string().min(1).max(64)` — prevents empty or oversized values |
| T-5-05 | Information Disclosure | session-snapshot broadcast | accept | All participants see all claim data and the finalized bill; this is by design (everyone at the table sees everyone's total) |
| T-5-06 | Denial of Service | server.ts finalize handler | accept | A malicious participant could spam finalize messages, but the idempotency guard means only the first succeeds; subsequent messages are no-ops |
</threat_model>

    import { useState } from 'react'
    import CameraCapture from '@/components/host/CameraCapture'
    import OcrReview from '@/components/host/OcrReview'
    import ShareScreen from '@/components/host/ShareScreen'
    import type { OcrResult } from '@/types'

    type Screen = 'capture' | 'reviewing' | 'share'

    export default function HostPage() {
      const [screen, setScreen] = useState<Screen>('capture')
      const [ocrResult, setOcrResult] = useState<OcrResult | null>(null)
      const [sessionId, setSessionId] = useState<string | null>(null)

      if (screen === 'capture') {
        return (
          <CameraCapture
            onComplete={(result) => {
              setOcrResult(result)
              setScreen('reviewing')
            }}
          />
        )
      }
      if (screen === 'reviewing') {
        return (
          <OcrReview
            initial={ocrResult!}
            onComplete={(id) => {
              setSessionId(id)
              setScreen('share')
            }}
          />
        )
      }
      return <ShareScreen sessionId={sessionId!} />
    }

  <files>components/host/CameraCapture.tsx</files>
  <action>
    Create `components/host/CameraCapture.tsx` as a `'use client'` component. Implement the full capture → preview → submit → OCR flow.

    Component interface:
    ```
    interface Props {
      onComplete: (result: OcrResult) => void
    }
    ```

    State variables:
    - `previewUrl: string | null` — blob URL from URL.createObjectURL(file)
    - `capturedFile: File | null` — the File from the input
    - `error: string | null` — error message shown in banner
    - `isPending: boolean` — from useTransition, true while OCR is in-flight

    File input: hidden (`className="sr-only"`), triggered by a styled "Take Photo" button. Use `useRef<HTMLInputElement>` to call `inputRef.current?.click()`.

    ```
    <input
      ref={inputRef}
      type="file"
      accept="image/*"
      capture="environment"
      className="sr-only"
      onChange={handleFileChange}
    />
    ```

    `handleFileChange`: revoke any previous previewUrl, create new blob URL via `URL.createObjectURL(file)`, set state.

    `handleRetake`: revoke previewUrl, reset previewUrl and capturedFile to null, reset input value with `inputRef.current.value = ''` (CRITICAL: without this, re-selecting the same file won't fire onChange again — see RESEARCH.md Pitfall 5).

    Canvas compression helper (inline function, not a separate module):
    ```
    async function compressImage(file: File): Promise<Blob> {
      const bitmap = await createImageBitmap(file)
      const maxDim = 2000
      const scale = Math.min(1, maxDim / Math.max(bitmap.width, bitmap.height))
      const canvas = document.createElement('canvas')
      canvas.width = Math.round(bitmap.width * scale)
      canvas.height = Math.round(bitmap.height * scale)
      canvas.getContext('2d')!.drawImage(bitmap, 0, 0, canvas.width, canvas.height)
      return new Promise(resolve => canvas.toBlob(b => resolve(b!), 'image/jpeg', 0.7))
    }
    ```

    `handleSubmit` (wrapped in `startTransition`):
    1. Call `compressImage(capturedFile)`
    2. Append to FormData as `formData.append('image', compressed, 'receipt.jpg')`
    3. POST to `/api/ocr`
    4. On success: revoke previewUrl, call `onComplete(result)`
    5. On HTTP error OR catch: set error message "Could not read the receipt. You can add items manually instead.", call `onComplete({ items: [], taxCents: 0, tipCents: 0 })` — this satisfies OCR-04 by letting the host proceed to an empty correction screen.

    UI layout (Tailwind v4 classes):
    - Full-screen column, safe padding, centered
    - "Take Photo" button: full-width, prominent color (e.g. `bg-blue-600 text-white`)
    - If previewUrl: show `<img src={previewUrl} alt="Receipt preview" className="w-full rounded-lg mt-4" />`
    - Below preview: two buttons in a row — "Retake" (secondary/outline) and "Submit Receipt" (primary, disabled and showing "Analyzing..." when isPending)
    - Error banner: shown when `error` is set — amber background, text message, "Continue anyway" button that calls `onComplete({ items: [], taxCents: 0, tipCents: 0 })`

    Cleanup: Call `URL.revokeObjectURL(previewUrl)` in the submit handler after advancing (see pattern in RESEARCH.md). Do NOT rely on unmount cleanup alone.
